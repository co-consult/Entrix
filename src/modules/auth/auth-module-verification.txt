class MfaMappingVerification {
  /**
   * Vérifie que le mapping Provider ↔ Method fonctionne
   */
  static verifyMapping() {
    console.log('🔍 Vérification du mapping MFA...');
    
    // Test mapping Provider → Method
    const providerToMethodTests = [
      { provider: 'SMS_OTP' as MfaProvider, expected: 'SMS' },
      { provider: 'EMAIL_OTP' as MfaProvider, expected: 'EMAIL' },
      { provider: 'TOTP_APP' as MfaProvider, expected: 'TOTP' },
      { provider: 'BACKUP_CODE' as MfaProvider, expected: 'BACKUP_CODES' }
    ];
    
    providerToMethodTests.forEach(test => {
      const method = MFA_PROVIDER_TO_METHOD_MAPPING[test.provider];
      console.assert(method === test.expected, 
        `❌ Provider mapping: ${test.provider} → ${method} (expected: ${test.expected})`);
    });
    
    // Test mapping Method → Provider
    const methodToProviderTests = [
      { method: 'SMS', expected: 'SMS_OTP' },
      { method: 'EMAIL', expected: 'EMAIL_OTP' },
      { method: 'TOTP', expected: 'TOTP_APP' },
      { method: 'BACKUP_CODES', expected: 'BACKUP_CODE' }
    ];
    
    methodToProviderTests.forEach(test => {
      const provider = MFA_METHOD_TO_PROVIDER_MAPPING[test.method as keyof typeof MFA_METHOD_TO_PROVIDER_MAPPING];
      console.assert(provider === test.expected, 
        `❌ Method mapping: ${test.method} → ${provider} (expected: ${test.expected})`);
    });
    
    console.log('✅ Mapping MFA vérifié avec succès');
  }

  /**
   * Vérifie que les validations fonctionnent
   */
  static verifyValidation() {
    console.log('🔍 Vérification des validations...');
    
    
    
    // Test validation SMS code
    console.assert(MFA_VALIDATION.isValidSmsCode('123456') === true, '❌ SMS code validation');
    console.assert(MFA_VALIDATION.isValidSmsCode('12345') === false, '❌ SMS code validation (invalid)');
    console.assert(MFA_VALIDATION.isValidSmsCode('12345a') === false, '❌ SMS code validation (letters)');
    
    // Test validation email code
    console.assert(MFA_VALIDATION.isValidEmailCode('123456') === true, '❌ Email code validation');
    console.assert(MFA_VALIDATION.isValidEmailCode('1234567') === false, '❌ Email code validation (too long)');
    
    // Test validation TOTP code
    console.assert(MFA_VALIDATION.isValidTotpCode('123456') === true, '❌ TOTP code validation');
    console.assert(MFA_VALIDATION.isValidTotpCode('abc123') === false, '❌ TOTP code validation (letters)');
    
    // Test validation backup code
    console.assert(MFA_VALIDATION.isValidBackupCode('ABCD-1234') === true, '❌ Backup code validation');
    console.assert(MFA_VALIDATION.isValidBackupCode('abcd-1234') === false, '❌ Backup code validation (lowercase)');
    console.assert(MFA_VALIDATION.isValidBackupCode('ABCD1234') === false, '❌ Backup code validation (no dash)');
    
    // Test format backup code
    const formatted = MFA_VALIDATION.formatBackupCode('ABCD1234');
    console.assert(formatted === 'ABCD-1234', '❌ Backup code formatting');
    
    // Test mask phone
    const maskedPhone = MFA_VALIDATION.maskPhoneNumber('+21650560560');
    console.assert(maskedPhone.includes('*'), '❌ Phone masking');
    console.assert(maskedPhone.startsWith('+216'), '❌ Phone masking (prefix)');
    
    // Test mask email
    const maskedEmail = MFA_VALIDATION.maskEmail('test@entrix.tn');
    console.assert(maskedEmail.includes('*'), '❌ Email masking');
    console.assert(maskedEmail.endsWith('@entrix.tn'), '❌ Email masking (domain)');
    
    console.log('✅ Validations vérifiées avec succès');
  }
}
